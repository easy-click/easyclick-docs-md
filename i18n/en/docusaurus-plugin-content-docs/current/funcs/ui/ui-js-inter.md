## Overview

- This chapter covers interaction between JS and UI elements
- JS modules that interact with UI are prefixed with `ui`, e.g. `ui.toast("msg")`

## How to Use

- In your project's `layout` folder, create a `ui.js` file with the following content:

```javascript showLineNumbers
function main() {
    ui.toast("UI controlled by JS");
}

main();
```

- Load and display `main.xml` from the layout folder:

```javascript showLineNumbers
function main() {
    ui.toast("UI controlled by JS");
    ui.layout("Tab 1", "main.xml");
}

main();
```

- This displays the `main.xml` layout in a simple UI

## Advanced Usage

- Example `ui.js`:

```javascript showLineNumbers
/**
 * Auto-generated by EasyClick development tools
 */
function main() {
    ui.toast("UI toast function");
    var set = ui.layout("Parameter Settings", "main.xml");
    ui.layout("Other Notes", "main2.xml");
    ui.logd("Set UI result: " + set);

    // Switch toggle usage
    var auto_env = ui.getViewValue(ui.auto_env);
    ui.logd("Value of tag auto_env: " + auto_env);
    // Switch checkedChange event
    ui.setEvent(ui.auto_env, "checkedChange", function (view, isChecked) {
        ui.logd("tag auto_env isChecked " + isChecked);
        if (isChecked) {
            startAutoEnv();
        }
    });
    if (ui.isServiceOk()) {
        ui.auto_env.setChecked(true);
    } else {
        ui.auto_env.setChecked(false);
    }

    // EditText usage
    var name = ui.getViewValue(ui.name);
    ui.logd("Value of tag name: " + name);
    ui.name.setText("name field value");
    // Spinner usage
    var sex = ui.getViewValue(ui.sex);
    ui.logd("Value of tag sex: " + sex);
    // Spinner itemSelected event
    ui.setEvent(ui.sex, "itemSelected", function (position, value) {
        ui.logd("tag sex itemSelected " + value);
    });

    // RadioButton usage
    var three = ui.getViewValue(ui.three);
    ui.logd("Value of tag three: " + three);
    // RadioButton checkedChange event
    ui.setEvent(ui.three, "checkedChange", function (view, isChecked) {
        ui.logd("tag three isChecked " + isChecked);
    });
    // CheckBox usage
    var dance = ui.getViewValue(ui.dance);
    ui.logd("Value of tag dance: " + dance);
    // CheckBox checkedChange event
    ui.setEvent(ui.dance, "checkedChange", function (view, isChecked) {
        ui.logd("tag dance isChecked " + isChecked);
    });

    // saveAllBtn save parameters click event
    ui.setEvent(ui.saveAllBtn, "click", function (view) {
        var s = ui.saveAllConfig();
        ui.logd("Save all parameters result " + s)
    });

    // System settings button
    ui.setEvent(ui.systemSetting, "click", function (view) {
        ui.openECSystemSetting();

    });
    // Start script button
    ui.setEvent(ui.startBtn, "click", function (view) {
        ui.start();
    });
    // Start environment button
    ui.setEvent(ui.envBtn, "click", function (view) {
        // Start environment async; on success update auto_env switch
        startAutoEnv();
    });
    // Get all UI parameters
    ui.logd("All UI parameters: " + ui.getConfigJSON());
    // setViewValue examples (commented out)
    // ui.setViewValue(ui.name, "set value");
    // ui.setViewValue(ui.auto_env, false);
    // ui.setViewValue(ui.sex, "Male|Female");
    // ui.setViewValue(ui.three, true);
    // ui.setViewValue(ui.dance, false)
    // Temporary shared storage for variables and data
    ui.putShareData("nameVar", ui.name);
    ui.putShareData("value", "my value");


}

function startAutoEnv() {
    ui.startEnvAsync(function (r) {
        ui.logd("Start environment result: " + r);
        ui.auto_env.setChecked(r);
    });
}

main();
```

## How Scripts Interact with JS

- The following code runs in `js/main.js`:

```javascript showLineNumbers
function main() {
    // Script reads UI objects — UI must be loaded first
    // Reference UI objects directly
    // Assume xml has tag=name EditText
    // Read variables/data set in the UI module
    logd(ui.getShareData("nameVar"))
    logd(ui.getShareData("value"))


    // Method 1: variable reference
    // Reset JS variables
    ui.resetUIVar()
    // Log ui.name
    logd("ui.name " + ui.name)
    if (ui.name) {
        // Run on UI main thread
        getHandler().post(function () {
            ui.name.setText("Fsdafsadfsa")
            ui.logd("EditText content: " + ui.name.getEditableText());
        });

    }
    sleep(2000);
    // Method 2: findViewByTag
    var name = ui.findViewByTag("name");
    if (name) {
        // Run on UI main thread
        getHandler().post(function () {
            name.setText("updated text")
            ui.logd("EditText content: " + name.getEditableText());
        });

    }


    sleep(5000);

}

main();
```

```javascript showLineNumbers
        ui.logd("tag dance isChecked " + isChecked);
    });
    ui.setEvent(ui.startBtn, "click", function (view) {
        ui.start();
    });
```

```javascript showLineNumbers
main();
```

## Dialogs

### alert

* Alert dialog
* @param map e.g. ```{"title":""}``` — fixed keys:
* title — dialog title
* msg — message
* cancelText — cancel button text
* okText — OK button text
* cancelable — whether dismissible
* @param okBtnCallback OK button callback
* @param cancelBtnCallback cancel button callback
* @param dismissListener dialog dismiss callback
* @return boolean true = success, false = failure

```javascript showLineNumbers
function main() {
 var p = {
 "title": "Confirm",
 "msg": "Are you sure?",
 "cancelable": true,
 "cancelText": "No",
 "okText": "Yes"
 };
 ui.alert(p,
 function (dialog, v) {
 logd("ok btn click : ")
 // Dismiss dialog
 dialog.doDismiss();
 return true;
 },
 function (dialog, v) {
 logd("cancel btn click ")
 // Dismiss dialog
 dialog.doDismiss();
 return true;
 },
 function () {
 logd("mission btn click ")
 return true;
 });

}

main();
```


### setAgentSupportNode Set Node Retrieval Mode in Agent Mode

* Set node retrieval mode in agent mode
* Only applies in agent mode
* EC Android 11.2.0+
* Call before starting agent service — modes 2 and 3 reduce detection signatures
* Mode 1 may trigger ruru detection of AccessibilityManager.isEnabled; modes 2+ do not
